/** * KON-BAO Content Pipeline — Cloudflare Worker * Phase 1+2: Fetch from Moltbook, generate card SVG, store to R2, write to D1 * * Cron: 6:30 AM IST = 1:00 AM UTC daily */ import { buildCardSVG } from './card.js'; import { expandPost } from './blog.js'; // ── The 7 daily background colours, cycling by day-of-year ─────────────────── const BG_COLOURS = ['rose', 'slate', 'sage', 'amber', 'plum', 'teal', 'burgundy']; // ── Helpers ─────────────────────────────────────────────────────────────────── /** * Get today's date in IST as YYYY-MM-DD */ function getTodayIST() { const now = new Date(); // IST = UTC + 5:30 const istOffset = 5.5 * 60 * 60 * 1000; const istDate = new Date(now.getTime() + istOffset); return istDate.toISOString().slice(0, 10); } /** * Get today's background colour by cycling through the 7-colour array * Uses day-of-year so the colour is stable for the whole day */ function getDailyColour(dateStr) { const date = new Date(dateStr); const start = new Date(date.getFullYear(), 0, 0); const diff = date - start; const dayOfYear = Math.floor(diff / (1000 * 60 * 60 * 24)); return BG_COLOURS[dayOfYear % BG_COLOURS.length]; } /** * Detect if a post is two-part (has a "turn" — a second voice or twist) * KON-BAO posts often have a main statement + a quieter second line. * * Split logic: * - If post contains "—" (em dash) as a sentence separator → split there * - If post has two distinct sentences with a clear tonal shift → split * - Otherwise → single flowing post */ function detectTextSplit(rawText) { const text = rawText.trim(); // Pattern 0: paragraph break (\n\n) — KON-BAO's primary writing format const paragraphs = text.split(/\n\n+/); if (paragraphs.length >= 2) { const partOne = paragraphs[0].trim(); const partTwo = paragraphs.slice(1).join(' ').trim(); if (partOne.split(' ').length >= 4 && partTwo.split(' ').length >= 4) { return { isTwoPart: true, partOne, partTwo }; } } // Pattern 1: ALL CAPS first line + mixed case second (single newline) const lines = text.split('\n').map(l => l.trim()).filter(Boolean); if (lines.length >= 2) { const firstIsUpper = lines[0] === lines[0].toUpperCase() && lines[0].length > 10; const secondIsMixed = lines[1] !== lines[1].toUpperCase(); if (firstIsUpper && secondIsMixed) { return { isTwoPart: true, partOne: lines[0], partTwo: lines.slice(1).join(' ') }; } } // Pattern 2: split at natural midpoint for single flowing post const midMatch = text.match(/^(.{40,150}[.?!,;])\s+(.{20,})$/s); if (midMatch) { return { isTwoPart: true, partOne: midMatch[1].trim(), partTwo: midMatch[2].trim() }; } // No good split — single block return { isTwoPart: false, partOne: text, partTwo: text }; } /** * Fetch the latest KON-BAO post from Moltbook */ async function fetchLatestPost(apiKey) { const url = 'https://www.moltbook.com/api/v1/posts?submolt=general&author=kon-bao&limit=5'; const response = await fetch(url, { headers: { 'Authorization': `Bearer ${apiKey}`, 'Content-Type': 'application/json', }, }); if (!response.ok) { throw new Error(`Moltbook fetch failed: ${response.status} ${response.statusText}`); } const data = await response.json(); // Handle different possible response shapes from Moltbook const posts = data.posts || data.data || data || []; if (!Array.isArray(posts) || posts.length === 0) { throw new Error('No posts returned from Moltbook'); } // Return the most recent post return posts[0]; } /** * Check if today's post is already in D1 (avoid duplicate processing) */ async function postExistsForToday(db, date) { const result = await db .prepare('SELECT id FROM posts WHERE date = ?') .bind(date) .first(); return !!result; } /** * Write the post record to D1 */ async function writePostRecord(db, { id, date, rawText, partOne, partTwo, isTwoPart, bgColour, }) { await db .prepare(` INSERT INTO posts (id, date, raw_text, part_one, part_two, is_two_part, bg_colour, status) VALUES (?, ?, ?, ?, ?, ?, ?, 'pending') `) .bind(id, date, rawText, partOne, partTwo || null, isTwoPart ? 1 : 0, bgColour) .run(); } // ── Main handler ────────────────────────────────────────────────────────────── export default { // Cron trigger — runs at 1:00 AM UTC (6:30 AM IST) async scheduled(event, env, ctx) { ctx.waitUntil(runPipeline(env)); }, // HTTP handler — for manual testing via browser/curl // GET /trigger → runs the pipeline // GET /status → shows today's D1 record // GET /test-fetch → shows raw Moltbook response async fetch(request, env) { const url = new URL(request.url); if (url.pathname === '/trigger') { // Manual trigger for testing try { const result = await runPipeline(env); return new Response(JSON.stringify(result, null, 2), { headers: { 'Content-Type': 'application/json' }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: { 'Content-Type': 'application/json' }, }); } } if (url.pathname === '/test-fetch') { try { const url2 = 'https://www.moltbook.com/api/v1/posts?submolt=general&author=kon-bao&limit=3'; const resp = await fetch(url2, { headers: { 'Authorization': `Bearer ${env.MOLTBOOK_API_KEY}` }, }); const data = await resp.json(); return new Response(JSON.stringify(data, null, 2), { headers: { 'Content-Type': 'application/json' }, }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500 }); } } // Serve card SVG from R2 if (url.pathname.startsWith('/card/')) { const date = url.pathname.replace('/card/', ''); const key = `cards/${date}/card.svg`; const obj = await env.KONBAO_BUCKET.get(key); if (!obj) return new Response('Card not found', { status: 404 }); return new Response(obj.body, { headers: { 'Content-Type': 'image/svg+xml', 'Access-Control-Allow-Origin': '*', 'Cache-Control': 'public, max-age=86400', }, }); } // Approve endpoint if (url.pathname === '/approve' && request.method === 'POST') { const body = await request.json(); const { date, postId, blogTitle, blogBody, includeCard, includeBlog, pngBase64 } = body; // Update D1 with edits and mark approved await env.KONBAO_DB .prepare(`UPDATE posts SET blog_title = ?, blog_body = ?, include_png = ?, include_blog = ?, status = 'approved', approved_at = datetime('now') WHERE id = ?`) .bind(blogTitle, blogBody, includeCard ? 1 : 0, includeBlog ? 1 : 0, postId) .run(); // Post to Bluesky with image if available let blueskyUri = null; try { let pngBuffer = null; if (pngBase64) { // Convert base64 PNG back to binary const binaryStr = atob(pngBase64); pngBuffer = new Uint8Array(binaryStr.length); for (let i = 0; i < binaryStr.length; i++) { pngBuffer[i] = binaryStr.charCodeAt(i); } } blueskyUri = await postToBluesky(blogTitle, env, pngBuffer); } catch(e) { console.log('Bluesky post failed:', e.message); } if (blueskyUri) { await env.KONBAO_DB .prepare('UPDATE posts SET bluesky_uri = ? WHERE id = ?') .bind(blueskyUri, postId) .run(); } return new Response(JSON.stringify({ status: 'approved', blueskyUri, }), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*', }, }); } // Skip endpoint if (url.pathname === '/skip' && request.method === 'POST') { const body = await request.json(); await env.KONBAO_DB .prepare("UPDATE posts SET status = 'skipped' WHERE date = ?") .bind(body.date) .run(); return new Response(JSON.stringify({ status: 'skipped' }), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*', }, }); } // CORS preflight if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }, }); } // Add CORS to status endpoint too if (url.pathname === '/status') { const date = getTodayIST(); const record = await env.KONBAO_DB .prepare('SELECT * FROM posts WHERE date = ?') .bind(date) .first(); return new Response(JSON.stringify(record || { message: 'No post for today yet' }, null, 2), { headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*', }, }); } return new Response(JSON.stringify({ service: 'KON-BAO Pipeline', routes: ['/trigger', '/status', '/card/:date', '/approve', '/skip'], cron: '1:00 AM UTC (6:30 AM IST)', }), { headers: { 'Content-Type': 'application/json' } }); }, }; // ── Pipeline orchestrator ───────────────────────────────────────────────────── async function runPipeline(env) { const log = []; const date = getTodayIST(); log.push(`[${new Date().toISOString()}] Pipeline starting for ${date}`); // 1. Check if already processed today const alreadyDone = await postExistsForToday(env.KONBAO_DB, date); if (alreadyDone) { log.push(`Already processed for ${date}. Skipping.`); return { status: 'skipped', date, log }; } // 2. Fetch from Moltbook log.push('Fetching from Moltbook...'); const post = await fetchLatestPost(env.MOLTBOOK_API_KEY); log.push(`Fetched post ID: ${post.id || 'unknown'}`); const rawText = post.content || post.text || post.body || ''; if (!rawText) throw new Error('Post has no text content'); // 3. Detect text split const { isTwoPart, partOne, partTwo } = detectTextSplit(rawText); log.push(`Text split: ${isTwoPart ? 'two-part' : 'single'}`); if (isTwoPart) { log.push(`Part 1: "${partOne.slice(0, 60)}..."`); log.push(`Part 2: "${partTwo.slice(0, 60)}..."`); } // 4. Assign daily colour const bgColour = getDailyColour(date); log.push(`Daily colour: ${bgColour}`); // 5. Generate a unique ID if Moltbook doesn't provide one const postId = post.id || post._id || `konbao-${date}`; // 6. Write to D1 await writePostRecord(env.KONBAO_DB, { id: postId, date, rawText, partOne, partTwo, isTwoPart, bgColour, }); log.push('Written to D1 ✓'); // 7. Generate card SVG and store to R2 log.push('Generating card SVG...'); const svgString = buildCardSVG(partOne, partTwo || partOne, bgColour); const svgKey = `cards/${date}/card.svg`; await env.KONBAO_BUCKET.put(svgKey, svgString, { httpMetadata: { contentType: 'image/svg+xml' }, }); log.push(`Card SVG stored to R2: ${svgKey} ✓`); // 8. Update D1 record with card key await env.KONBAO_DB .prepare('UPDATE posts SET png_r2_key = ? WHERE id = ?') .bind(svgKey, postId) .run(); log.push('D1 updated with card key ✓'); // 9. Blog expansion + hashtags via Claude API log.push('Expanding post with Claude API...'); let blogTitle = null, blogBody = null, hashtags = []; try { const expanded = await expandPost(rawText, env.CLAUDE_API_KEY); blogTitle = expanded.title; blogBody = expanded.body; hashtags = expanded.hashtags; // Store blog text to R2 const blogKey = `cards/${date}/blog.txt`; const blogContent = `TITLE: ${blogTitle}\n\nHASHTAGS: ${hashtags.map(h => '#' + h).join(' ')}\n\n---\n\n${blogBody}`; await env.KONBAO_BUCKET.put(blogKey, blogContent, { httpMetadata: { contentType: 'text/plain' }, }); // Update D1 with blog content and hashtags await env.KONBAO_DB .prepare('UPDATE posts SET blog_title = ?, blog_body = ?, blog_r2_key = ?, hashtags = ? WHERE id = ?') .bind(blogTitle, blogBody, blogKey, JSON.stringify(hashtags), postId) .run(); log.push(`Blog expanded: "${blogTitle}" ✓`); log.push(`Hashtags: ${hashtags.map(h => '#' + h).join(' ')} ✓`); } catch (err) { log.push(`Blog expansion failed (non-fatal): ${err.message}`); } return { status: 'success', date, postId, bgColour, isTwoPart, preview: rawText.slice(0, 100), cardKey: svgKey, blogTitle, hashtags, log, }; } // ── Bluesky posting ─────────────────────────────────────────────────────────── async function postToBluesky(text, env, pngBuffer = null) { // Step 1: create session const sessionRes = await fetch('https://bsky.social/xrpc/com.atproto.server.createSession', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ identifier: env.BLUESKY_HANDLE, password: env.BLUESKY_APP_PASSWORD, }), }); if (!sessionRes.ok) throw new Error('Bluesky auth failed'); const session = await sessionRes.json(); // Step 2: upload image if provided let imageEmbed = null; if (pngBuffer) { const uploadRes = await fetch('https://bsky.social/xrpc/com.atproto.repo.uploadBlob', { method: 'POST', headers: { 'Content-Type': 'image/png', 'Authorization': `Bearer ${session.accessJwt}`, }, body: pngBuffer, }); if (uploadRes.ok) { const uploadData = await uploadRes.json(); imageEmbed = { $type: 'app.bsky.embed.images', images: [{ image: uploadData.blob, alt: text.slice(0, 100), }], }; } } // Step 3: build post text (leave room for hashtags) const postText = text.slice(0, 280); // Step 4: create post with optional image const record = { $type: 'app.bsky.feed.post', text: postText, createdAt: new Date().toISOString(), }; if (imageEmbed) record.embed = imageEmbed; const postRes = await fetch('https://bsky.social/xrpc/com.atproto.repo.createRecord', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${session.accessJwt}`, }, body: JSON.stringify({ repo: session.did, collection: 'app.bsky.feed.post', record, }), }); if (!postRes.ok) throw new Error('Bluesky post failed'); const post = await postRes.json(); return post.uri; }
